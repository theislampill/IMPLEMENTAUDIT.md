import hashlib
import json
from pathlib import Path
import sys

mode = sys.argv[1]
subject = Path(sys.argv[2])
data = subject.read_bytes()
digest = hashlib.sha256(data).hexdigest()
if mode == "policy":
    assert digest == "3debfa7f66e5b97df1f44a19e28ba39cdef4a343ad3ea8791e108a782146fed4"
    value = json.loads(data)
    assert value["schema"] == "implementaudit.codex-stop-policy.v1"
    assert value["session_id"] == "00000000-0000-0000-0000-000000000000"
    result = {"scope": "independent policy byte pin", "source_sha256": digest, "bytes": len(data)}
else:
    assert mode == "transport"
    echo_file = Path(sys.argv[3])
    raw = echo_file.read_bytes()
    proof = json.loads(raw)
    cases = {item["case"]: item for item in proof["rows"] if "case" in item}
    assert len(cases) == len(proof["rows"]) == 7
    for key in ("stock-codepage-437", "explicit-utf8-no-bom", "guarded-correction"):
        row = cases[key]
        assert bytes.fromhex(row["received_hex"]) == bytes.fromhex(row["accounted_hex"]) == data
        assert row["exact_match"] is True
    old = cases["stock-codepage-65001"]
    assert bytes.fromhex(old["received_hex"]) == b"\xef\xbb\xbf" + data
    zero = cases["stock-codepage-65001-no-write"]
    assert zero["accounted_bytes"] == 0 and bytes.fromhex(zero["received_hex"]) == b"\xef\xbb\xbf"
    result = {"scope": "local raw byte transport under corrected private console guard", "source_sha256": digest,
              "bytes": len(data), "echo_sha256": hashlib.sha256(raw).hexdigest(), "uncorrected_extra_bytes": 3,
              "guarded_exact": True}
print(json.dumps(result, sort_keys=True))
