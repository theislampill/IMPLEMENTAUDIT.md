function Need($ok,$why){if(!$ok){throw $why}}
function Physical($path,$directory){
 $item=Get-Item -LiteralPath $path -Force
 Need (($item -is [IO.DirectoryInfo]) -eq $directory) 'path type'
 for($at=$item;$null -ne $at;){
  Need (-not ($at.Attributes -band [IO.FileAttributes]::ReparsePoint)) 'aliased path'
  if($at -is [IO.DirectoryInfo]){$at=$at.Parent}else{$at=$at.Directory}
 }
}

function Offline{
 $loop=@()
 $nic=@([Net.NetworkInformation.NetworkInterface]::GetAllNetworkInterfaces())
 foreach($adapter in $nic){
  if($adapter.NetworkInterfaceType -eq 'Loopback'){
   $ip=$adapter.GetIPProperties()
   if($adapter.Supports('IPv4')){$loop+=$ip.GetIPv4Properties().Index}
   if($adapter.Supports('IPv6')){$loop+=$ip.GetIPv6Properties().Index}
  }else{
   Need ($adapter.OperationalStatus -in @('Down','NotPresent','LowerLayerDown')) 'nonloopback NIC active'
  }
 }
 $routes=@(Get-NetRoute -PolicyStore ActiveStore -ErrorAction Stop)
 Need ($loop.Count -gt 0) 'loopback identity absent'
 foreach($route in $routes){Need ($route.InterfaceIndex -in $loop) 'nonloopback active route'}
 @{adapters=$nic.Count;loopback=$loop;routes=$routes.Count}
}

function Members($node){@($node.SelectNodes('*'))}
function Text-Member($node,$name,$kind){
 $x=$node.SelectSingleNode($name)
 Need ($null -ne $x -and $x.GetAttribute('type') -ceq $kind) 'invalid member type'

}
function Array-Member($node,$name){
 $x=$node.SelectSingleNode($name)
 Need ($null -ne $x -and $x.GetAttribute('type') -ceq 'array') 'expected array'
 Members $x
}
function Reply([byte[]]$bytes,[int]$id){
 $q=[Xml.XmlDictionaryReaderQuotas]::new();$q.MaxDepth=24;$q.MaxStringContentLength=131072
 $r=[Runtime.Serialization.Json.JsonReaderWriterFactory]::CreateJsonReader($bytes,$q)
 try{
  $doc=[Xml.XmlDocument]::new();$doc.XmlResolver=$null;$doc.Load($r)
  foreach($object in $doc.SelectNodes('//*[@type="object"]')){
   $names=[Collections.Generic.HashSet[string]]::new([StringComparer]::OrdinalIgnoreCase)
   foreach($x in (Members $object)){Need ($x.NamespaceURI -eq '' -and $names.Add($x.Name)) 'duplicate or foreign name'}
  }
  $x=$doc.DocumentElement
  Need ($x.GetAttribute('type') -ceq 'object' -and @(Members $x).Count -eq 2) 'foreign envelope'
  Need ((Text-Member $x 'id' 'number') -ceq [string]$id) 'foreign response ID'
  $result=$x.SelectSingleNode('result')
  Need ($null -ne $result -and $result.GetAttribute('type') -ceq 'object') 'response error'
  ,$result
 }finally{$r.Dispose()}
}

function Keep($c,$index,[byte[]]$bytes,[int]$n){
 $count=[Math]::Min($n,131072-$c.Bytes)
 $c.Log[$index].Write($bytes,0,$count);$c.Bytes+=$count
 Need ($count -eq $n) 'raw capture limit'
}
function Time-Left($c){$c.End-$c.Clock.ElapsedMilliseconds}
function Read-Next($c,$i){$c.Pending[$i]=$c.Stream[$i].ReadAsync($c.Buffer[$i],0,4096)}
function Pump($c){
 Need ((Time-Left $c) -gt 0) 'deadline'
 for($i=0;$i -lt 2;$i++){
  $read=$c.Pending[$i]
  if($null -ne $read -and $read.IsCompleted){
   $n=$read.GetAwaiter().GetResult()
   if($n -eq 0){$c.Pending[$i]=$null;continue}
   Keep $c $i $c.Buffer[$i] $n
   Need ($i -eq 0) 'unexpected stderr'
   Read-Next $c $i
  }
 }
}
function Frame($c,[int]$id){
 while($true){
  Pump $c
  $log=$c.Log[0];$buffer=$log.GetBuffer()
  while($c.Scan -lt $log.Length){
   $at=$c.Scan;$c.Scan+=1
   if($buffer[$at] -eq 10){
    Need ($c.Scan -eq $log.Length) 'extra stdout'
    $size=$at-$c.Cursor;$frame=[byte[]]::new($size)
    [Array]::Copy($buffer,$c.Cursor,$frame,0,$size)
    $reply=Reply $frame $id
    $c.Map.Add(@{stream='stdout';offset=$c.Cursor;length=$size+1})
    $c.Cursor=$c.Scan
    return ,$reply
   }
  }
  Need ($null -ne $c.Pending[0]) 'stdout EOF before response'
  Start-Sleep -Milliseconds 5
 }
}
function Finish-Write($c,$task){
 while(!$task.IsCompleted){Pump $c;Start-Sleep -Milliseconds 5}
 [void]$task.GetAwaiter().GetResult()
}
function Send($c,[string]$wire){
 Pump $c
 Need ($c.Log[0].Length -eq $c.Cursor -and $null -ne $c.Pending[0]) 'unsolicited stdout or EOF'
 $data=$utf.GetBytes($wire+[char]10)
 Need ($c.Bytes+$data.Length -le 131072) 'send cap'
 $c.WritePending=$true
 Finish-Write $c ($c.In.WriteAsync($data,0,$data.Length))
 Finish-Write $c ($c.In.FlushAsync())
 $c.Map.Add(@{stream='stdin';offset=$c.Log[2].Length;length=$data.Length})
 Keep $c 2 $data $data.Length
 $c.WritePending=$false
 Need ((Time-Left $c) -gt 0) 'send deadline'
}

}
